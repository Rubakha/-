## Код TG бота:

```python
from datetime import datetime

def get_letter_price(letter_type):
    """Возвращает цену в зависимости от месяца"""
    
    current_date = datetime.now()
    is_september = current_date.month == 9
    
    prices = {
        "september": {
            "Письмо": 390,
            "Дневник": 590,
            "Сценарий": 890
        },
        "october_plus": {
            "Письмо": 690,
            "Дневник": 890,
            "Сценарий": 1290
        }
    }
    
    if is_september:
        return prices["september"].get(letter_type, 390)
    else:
        return prices["october_plus"].get(letter_type, 690)

# Использование:
price = get_letter_price("Письмо")  # 390 в сентябре, 690 в октябре
```
